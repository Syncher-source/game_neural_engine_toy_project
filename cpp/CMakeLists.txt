# ---- deterministic world model: the CUDA kernel source and the shared math header are embedded
# as byte arrays and compiled at runtime by NVRTC (no CUDA toolkit needed to build).
function(nge_embed_bytes in_path out_var)
  file(READ "${in_path}" _hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _bytes "${_hex}")
  set(${out_var} "${_bytes}" PARENT_SCOPE)
endfunction()
nge_embed_bytes("${CMAKE_SOURCE_DIR}/src/det_kernel.cu" NGE_DET_KERNEL_BYTES)
nge_embed_bytes("${CMAKE_SOURCE_DIR}/include/nge/det_math.h" NGE_DET_MATH_BYTES)
configure_file(cmake/det_sources.cpp.in "${CMAKE_BINARY_DIR}/generated/det_sources.cpp" @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/det_kernel.cu include/nge/det_math.h)

# ---- core library: state, protocol, net, client logic, neural sim (ONNX Runtime), deterministic sim, renderer
add_library(nge_core STATIC
  src/state.cpp src/protocol.cpp src/net.cpp src/client_core.cpp src/neural_sim.cpp src/neural_renderer.cpp
  src/det_sim.cpp "${CMAKE_BINARY_DIR}/generated/det_sources.cpp")
target_include_directories(nge_core PUBLIC include)
target_link_libraries(nge_core PUBLIC ${CMAKE_DL_LIBS})
# det_sim.cpp: AVX2+FMA for the SIMD dot products, no floating-point contraction anywhere else
if(MSVC)
  set_source_files_properties(src/det_sim.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2;/fp:precise")
else()
  set_source_files_properties(src/det_sim.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma;-ffp-contract=off")
endif()
if(MSVC)
  target_compile_options(nge_core PUBLIC /W3 /permissive- /utf-8)
  target_compile_definitions(nge_core PUBLIC _CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
  target_compile_options(nge_core PUBLIC -Wall -Wextra -Wno-unused-parameter)
endif()
if(WIN32)
  target_link_libraries(nge_core PUBLIC ws2_32)
endif()

# ---- ONNX Runtime (prebuilt package). If ONNXRUNTIME_DIR is empty, download it into cpp/third_party.
set(NGE_ORT_VERSION "1.31.0" CACHE STRING "ONNX Runtime release to download when ONNXRUNTIME_DIR is not set")
if(WIN32)
  set(NGE_ORT_PACKAGE "onnxruntime-win-x64-gpu_cuda12" CACHE STRING "ONNX Runtime package name (onnxruntime-win-x64-gpu_cuda12 | onnxruntime-win-x64-gpu_cuda13 | onnxruntime-win-x64)")
  set(_ort_ext "zip")
else()
  set(NGE_ORT_PACKAGE "onnxruntime-linux-x64" CACHE STRING "ONNX Runtime package name (onnxruntime-linux-x64 | onnxruntime-linux-x64-gpu_cuda12)")
  set(_ort_ext "tgz")
endif()
if(NOT ONNXRUNTIME_DIR)
  set(_ort_name "${NGE_ORT_PACKAGE}-${NGE_ORT_VERSION}")
  set(_ort_dir "${CMAKE_SOURCE_DIR}/third_party/${_ort_name}")
  if(NOT EXISTS "${_ort_dir}/include/onnxruntime_cxx_api.h")
    set(_ort_url "https://github.com/microsoft/onnxruntime/releases/download/v${NGE_ORT_VERSION}/${_ort_name}.${_ort_ext}")
    set(_ort_archive "${CMAKE_SOURCE_DIR}/third_party/${_ort_name}.${_ort_ext}")
    message(STATUS "Downloading ONNX Runtime ${_ort_url} (this can take a few minutes) ...")
    file(DOWNLOAD "${_ort_url}" "${_ort_archive}" SHOW_PROGRESS STATUS _st TLS_VERIFY ON)
    list(GET _st 0 _code)
    if(NOT _code EQUAL 0)
      list(GET _st 1 _msg)
      message(WARNING "ONNX Runtime download failed: ${_msg}. Set ONNXRUNTIME_DIR manually.")
      file(REMOVE "${_ort_archive}")
    else()
      file(ARCHIVE_EXTRACT INPUT "${_ort_archive}" DESTINATION "${CMAKE_SOURCE_DIR}/third_party")
      file(REMOVE "${_ort_archive}")
    endif()
  endif()
  if(EXISTS "${_ort_dir}/include/onnxruntime_cxx_api.h")
    set(ONNXRUNTIME_DIR "${_ort_dir}")
  endif()
endif()
set(NGE_HAVE_ORT 0)
if(ONNXRUNTIME_DIR)
  find_path(ORT_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_DIR}/include" "${ONNXRUNTIME_DIR}/build/native/include" NO_DEFAULT_PATH)
  find_library(ORT_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_DIR}/lib" "${ONNXRUNTIME_DIR}/runtimes/win-x64/native" NO_DEFAULT_PATH)
  if(ORT_INCLUDE_DIR AND ORT_LIBRARY)
    set(NGE_HAVE_ORT 1)
    target_include_directories(nge_core PUBLIC ${ORT_INCLUDE_DIR})
    target_link_libraries(nge_core PUBLIC ${ORT_LIBRARY})
    get_filename_component(ORT_LIB_DIR ${ORT_LIBRARY} DIRECTORY)
    if(WIN32)
      file(GLOB ORT_RUNTIME_FILES "${ORT_LIB_DIR}/*.dll")   # copied next to the executables post-build
    endif()
    message(STATUS "ONNX Runtime: ${ORT_LIBRARY}")
  else()
    message(WARNING "ONNXRUNTIME_DIR=${ONNXRUNTIME_DIR} but headers/library not found")
  endif()
endif()
target_compile_definitions(nge_core PUBLIC NGE_HAVE_ORT=${NGE_HAVE_ORT})
if(NOT NGE_HAVE_ORT)
  message(WARNING "ONNX Runtime not configured: neural sim disabled, server/clients fall back to the classic sim")
endif()

# ---- CUDA / cuDNN runtime libraries from NVIDIA's redistributables: Windows gets everything the
# ONNX Runtime CUDA provider needs plus cudart for the D3D12 interop; Linux only NVRTC (for the
# deterministic kernel; the ONNX Runtime CPU package is used there).
option(NGE_CUDA_REDIST "Download CUDA/cuDNN/NVRTC runtime libraries from NVIDIA's redistributables" ON)
set(NGE_HAVE_CUDART 0)
set(NGE_CUDA_DLLS "")
if(NGE_HAVE_ORT)
  include(cmake/NgeCudaRedist.cmake)
  if(WIN32)
    if(NGE_CUDART_INCLUDE_DIRS AND EXISTS "${NGE_CUDART_LIB}")
      set(NGE_HAVE_CUDART 1)
      message(STATUS "cudart: ${NGE_CUDART_LIB}")
    else()
      message(WARNING "cudart headers/import library not found: D3D12 zero-readback interop disabled")
    endif()
  endif()
endif()

# ---- runtime library deployment: ONNX Runtime + CUDA + cuDNN + NVRTC next to the executables
if(ORT_RUNTIME_FILES OR NGE_CUDA_DLLS)
  add_custom_target(nge_deploy_dlls
    COMMAND ${CMAKE_COMMAND} -E make_directory "${NGE_RUNTIME_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${ORT_RUNTIME_FILES} ${NGE_CUDA_DLLS} "${NGE_RUNTIME_DIR}"
    COMMENT "deploying runtime libraries (ONNX Runtime / CUDA / cuDNN / NVRTC) to ${NGE_RUNTIME_DIR}"
    VERBATIM)
endif()

function(nge_app name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE nge_core)
  if(NGE_WITH_CLASSIC_SIM)
    target_link_libraries(${name} PRIVATE nge_classic)
    target_compile_definitions(${name} PRIVATE NGE_WITH_CLASSIC_SIM=1)
  else()
    target_compile_definitions(${name} PRIVATE NGE_WITH_CLASSIC_SIM=0)
  endif()
  set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${NGE_RUNTIME_DIR}")
  if(TARGET nge_deploy_dlls)
    add_dependencies(${name} nge_deploy_dlls)
  endif()
  if(NGE_HAVE_ORT AND NOT WIN32)
    set_target_properties(${name} PROPERTIES BUILD_RPATH "${ORT_LIB_DIR}" INSTALL_RPATH "${ORT_LIB_DIR}")
  endif()
endfunction()

  find_package(SDL3 CONFIG)

    target_link_libraries(nge_client PRIVATE SDL3::SDL3)
    if(NGE_WITH_CLASSIC_RENDER)

      target_compile_definitions(nge_client PRIVATE NGE_WITH_CLASSIC_RENDER=1)
    else()
      target_compile_definitions(nge_client PRIVATE NGE_WITH_CLASSIC_RENDER=0)
    endif()
    target_compile_definitions(nge_client PRIVATE NGE_HAVE_CUDART=${NGE_HAVE_CUDART})

    message(WARNING "SDL3 not found: graphical client (nge_client) skipped. Install via vcpkg (vcpkg.json) or set -DNGE_WITH_SDL3=OFF")
  endif()
